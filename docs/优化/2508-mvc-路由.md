         mvc.RouterWs("/ws/chat", chatCtrl, "HandleChat")
         
         // 3. 使用 RouterWsWithUpgrader 带自定义配置
         upgrader := websocket.HertzUpgrader{
             ReadBufferSize: 2048,
             WriteBufferSize: 2048,
         }
         mvc.RouterWsWithUpgrader("/ws/game", gameCtrl, "HandleGame", upgrader)
         
         app.Run()
     }

     5. 关键实现要点

     1. 统一接口设计 - 与现有的 Router、RouterPrefix 等方法保持一致的风格
     2. 链式调用支持 - 所有方法返回 *App 支持链式调用
     3. 反射机制 - 使用反射调用控制器的 WebSocket 方法
     4. 升级器配置 - 支持默认和自定义升级器
     5. 错误处理 - 验证方法存在性和签名正确性

     6. 预期效果

     用户可以像使用普通路由一样简单地注册 WebSocket 路由：

     // 简洁的 API
     mvc.HandlerWs("/ws", wsHandler)
     mvc.RouterWs("/chat", controller, "Chat")
     mvc.RouterWsWithUpgrader("/game", controller, "Game", upgrader)

     // 链式调用
     app.HandlerWs("/ws1", handler1)
        .HandlerWs("/ws2", handler2)
        .RouterWs("/ws3", ctrl, "Method")

     这个实现将使 WebSocket 功能完全集成到 YYHertz MVC 框架中，提供统一、简洁的 API。
