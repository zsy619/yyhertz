💻 代码示例

控制器代码：

package controllers func (c *BeegoController) Index() error { data := map[string]any{ "UserName": "张三", "IsLoggedIn": true, "Users": []User{ {Name: "李四", Email: "lisi@example.com", IsActive: true}, {Name: "王五", Email: "wangwu@example.com", IsActive: false}, }, "Stats": c.GetPerformanceStats(), } return c.RenderTemplate("beego/index", data) }

模板渲染：

// 使用统一API result, err := view.UnifiedRender("beego/index", data) // 使用Beego引擎 result, err := view.GetBeegoEngine().RenderTemplate("beego/index", data) // 使用布局 result, err := view.UnifiedRenderWithLayout("beego/index", "layouts/app", data)
